# The deep learning layer — lessons 9 and 10 only.
#
# Installed on top of the core image, so that pulling it later downloads this
# layer and nothing else.
#
# tensorflow-cpu, not tensorflow: the plain package pulls the nvidia-* CUDA
# libraries as hard dependencies, roughly 450 MB of them, even on a machine with
# no GPU. Every lab in this course runs on CPU by design.
#
# ============================
# Why ml-dtypes is pinned here
# ============================
# It is a TensorFlow dependency, not something the course imports, and it is the
# reason this file exists rather than three lines in the Dockerfile. Resolved
# freshly, pip selects ml-dtypes 0.6.0, which requires numpy>=2.0.0 — and numpy 2
# breaks pandas, scipy and scikit-learn, all compiled here against numpy 1.x.
# The first build of this image did exactly that and could not import
# tensorflow at all.
#
# 0.5.4 is the version the previously published image shipped, alongside
# numpy 1.26.4 and keras 3.15.1, and is therefore the combination the course was
# actually verified against.
#
# The Dockerfile also passes requirements.txt as a pip constraints file, so a
# future dependency cannot quietly move numpy underneath the pins.

# One TensorFlow per architecture, at the same version. tensorflow-cpu has no
# Linux arm64 wheel, so Apple-silicon Macs get the plain package there - which
# since 2.14 pulls the CUDA libraries only through the `and-cuda` extra, so it
# stays CPU-only and small. On amd64 nothing changes.
tensorflow-cpu==2.21.0; platform_machine == "x86_64"
tensorflow==2.21.0; platform_machine == "aarch64"
keras==3.15.1
ml-dtypes==0.5.4

# TensorFlow's own dependencies that a fresh resolve moved in October 2026,
# pinned to what the August image shipped, for the reason given at the end of
# requirements.txt.
absl-py==2.5.0
grpcio==1.83.0
protobuf==7.36.0
wrapt==2.3.0
